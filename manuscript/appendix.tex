\section{Dirichlet aggregation and the moments of linear functionals}\label{app:dirichlet}

Let $\bpi\sim\Dir(\bkappa)$ on the $J$-simplex with $m=\sum_j\kappa_j$, and for $\cR\subseteq\{1,\dots,J\}$ write $\pi(\cR)=\sum_{j\in\cR}\pi_j$ and $\kappa(\cR)=\sum_{j\in\cR}\kappa_j$.

\begin{lemma}[Aggregation]\label{lem:agg}
For any partition $\cR_1,\dots,\cR_L$ of $\{1,\dots,J\}$, $\big(\pi(\cR_1),\dots,\pi(\cR_L)\big)\sim\Dir\big(\kappa(\cR_1),\dots,\kappa(\cR_L)\big)$. In particular $\pi(\cR)\sim\Beta\big(\kappa(\cR),m-\kappa(\cR)\big)$ for every $\cR$.
\end{lemma}

\begin{proof}
Write $\pi_j=G_j/\sum_{l}G_l$ with $G_1,\dots,G_J$ independent and $G_j\sim\mathrm{Gamma}(\kappa_j,1)$. Then $\sum_{j\in\cR_l}G_j\sim\mathrm{Gamma}(\kappa(\cR_l),1)$, the $L$ group sums are independent, and the vector of their proportions is Dirichlet with the group-sum parameters. The Beta statement is the case $L=2$.
\end{proof}

\begin{lemma}[Moments of linear functionals]\label{lem:mom}
Let $\hat\bpi=\bkappa/m$. For $\bc\in\mathbb R^J$, $\E[\bc^\top\bpi]=\bc^\top\hat\bpi$ and
\begin{equation*}
\Var[\bc^\top\bpi]=\frac{\sum_j c_j^2\hat\pi_j-\big(\sum_j c_j\hat\pi_j\big)^2}{m+1} .
\end{equation*}
\end{lemma}

\begin{proof}
The Dirichlet moments are $\E[\pi_j]=\hat\pi_j$ and $\Cov(\pi_j,\pi_l)=(\hat\pi_j\ind\{j=l\}-\hat\pi_j\hat\pi_l)/(m+1)$, and $\Var[\bc^\top\bpi]=\sum_{j,l}c_jc_l\Cov(\pi_j,\pi_l)$.
\end{proof}

Applied to the posterior $\Dir(\bkappa_a+\bx_a)$, whose mass is $m_a+n_1$ and whose mean is $\hat\bpi_a$, Lemma~\ref{lem:agg} gives \eqref{eq:beta}. Lemma~\ref{lem:mom} with $\bc(\bw)=\sum_k w_k\ind_{\cR_k}$ gives \eqref{eq:moments}, because $\sum_j c_j\hat\pi_{a,j}=\sum_k w_k\hat p_{a,k}$ and, since $\ind_{\cR_k}(j)\ind_{\cR_{k'}}(j)=\ind_{\cR_k\cap\cR_{k'}}(j)$, $\sum_j c_j^2\hat\pi_{a,j}=\sum_{k,k'}w_kw_{k'}\hat p_{a,kk'}$. The same calculation with $\bc=\ind_{\cR_k}$ and $\bc=\ind_{\cR_{k'}}$ gives the posterior covariance $\Cov(p_{a,k},p_{a,k'}\mid\bx_a)=(\hat p_{a,kk'}-\hat p_{a,k}\hat p_{a,k'})/(m_a+n_1+1)$.

\section{Proof of Theorem~\protect\thmtone}\label{app:proof}

Let $\mathcal F_1$ denote the $\sigma$-field generated by the stage-1 data and the external information used at the interim. The selected dose $a^\star$, the futility decision, $n_2$ and every $P^{(1)}_{a,k}$ are $\mathcal F_1$-measurable. A statistic $P$ with values in $[0,1]$ is a valid $p$-value for a hypothesis $H$ if $\Pr(P\le u)\le u$ for all $u\in[0,1]$ whenever $H$ is true. It is conditionally valid given $\mathcal F_1$ if $\Pr(P\le u\mid\mathcal F_1)\le u$ almost surely on the event that $H$ is true.

\begin{lemma}[Combination of valid $p$-values]\label{lem:comb}
Let $\psi_m$ be the Bonferroni combination, or the Simes combination applied to $p$-values that satisfy the Simes inequality. If $P_1,\dots,P_m$ are valid $p$-values for $H_1,\dots,H_m$, then $\psi_m(P_1,\dots,P_m)$ is a valid $p$-value for $\bigcap_iH_i$. If the $P_i$ are conditionally valid given $\mathcal F_1$ and, for Simes, satisfy the Simes inequality conditionally, then $\psi_m(P_1,\dots,P_m)$ is conditionally valid. Both combinations are non-decreasing in each argument.
\end{lemma}

\begin{proof}
For Bonferroni, $\Pr(m\min_iP_i\le u)\le\sum_i\Pr(P_i\le u/m)\le u$, and the same inequality holds for conditional probabilities. For Simes the statement is the Simes inequality \citep{simes1986}, proved by \citet{sarkar1998} for multivariate totally positive statistics and by \citet{benjamini2001} under positive regression dependence on the subset of true hypotheses. Applied to the conditional distribution given $\mathcal F_1$ it gives the conditional statement. Monotonicity holds because increasing an argument cannot decrease any order statistic.
\end{proof}

\begin{lemma}[Intersection--union $p$-value]\label{lem:iu}
If $P_1,\dots,P_K$ are valid $p$-values for $H_1,\dots,H_K$, then $Q=\max_k P_k$ is a valid $p$-value for $\bigcup_k H_k$.
\end{lemma}

\begin{proof}
If $\bigcup_kH_k$ is true, some $H_{k_0}$ is true, and $\Pr(Q\le u)\le\Pr(P_{k_0}\le u)\le u$.
\end{proof}

\begin{lemma}[Inverse-normal combination across stages]\label{lem:inv}
Let $P_1$ be $\mathcal F_1$-measurable and valid, and let $P_2$ be conditionally valid given $\mathcal F_1$. Then for $v_1^2+v_2^2=1$ and $v_2>0$, the combination \eqref{eq:comb} satisfies $\Pr\big(\cC(P_1,P_2)\le\alpha\big)\le\alpha$.
\end{lemma}

\begin{proof}
Put $Z_i=\Phi^{-1}(1-P_i)$, so that $\cC(P_1,P_2)\le\alpha$ if and only if $v_1Z_1+v_2Z_2\ge z_{1-\alpha}$. Conditional validity gives $\Pr(Z_2\ge t\mid\mathcal F_1)=\Pr\big(P_2\le1-\Phi(t)\mid\mathcal F_1\big)\le1-\Phi(t)$ for all $t$. Hence
\begin{equation*}
\Pr\big(\cC\le\alpha\big)=\E\Big[\Pr\Big(Z_2\ge\frac{z_{1-\alpha}-v_1Z_1}{v_2}\,\Big|\,\mathcal F_1\Big)\Big]\le\E\big[g(Z_1)\big],\qquad g(z)=1-\Phi\Big(\frac{z_{1-\alpha}-v_1z}{v_2}\Big),
\end{equation*}
with $g$ non-decreasing. Validity of $P_1$ gives $\Pr(Z_1\ge t)\le1-\Phi(t)$, so $Z_1$ is stochastically smaller than a standard normal variable $Z$. Therefore $\E[g(Z_1)]\le\E[g(Z)]=\Pr(v_1Z+v_2Z'\ge z_{1-\alpha})=\alpha$ for $Z'$ standard normal and independent of $Z$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:t1e}]
Fix a configuration $(\bpi_0,\bpi_1,\dots,\bpi_M)$ and let $B_0=\{(a,k):H_{a,k}\text{ true}\}$. This set is determined by the configuration alone. If it is empty there is nothing to prove. By the closure principle, the rejection of a true $H_{a,k}$ requires the rejection of every intersection that contains $(a,k)$, in particular of $H_{B_0}$. A false claim rejects some true $H_{a^\star,k}$ and is such a rejection. It therefore suffices to show that $\Pr(H_{B_0}\text{ rejected})\le\alpha$.

A trial that stops for futility rejects nothing, so the rejection event is contained in the corresponding event for the trial that always proceeds to stage 2. We work with the latter, in which stage-2 data exist on every sample path and (i) and (iii) hold as stated. By \eqref{eq:rejJ}, $H_{B_0}$ is rejected if and only if $\cC\big(P^{(1)}_{B_0},P^{(2)}_{B_0}\big)\le\alpha$, with
\begin{equation*}
P^{(1)}_{B_0}=\psi_{|B_0|}\big((P^{(1)}_{a,k})_{(a,k)\in B_0}\big),\qquad
P^{(2)}_{B_0}=\psi_{|B_0^\star|}\big((P^{(2)}_k)_{k\in B_0^\star}\big),\quad B_0^\star=\{k:(a^\star,k)\in B_0\},
\end{equation*}
and $P^{(2)}_{B_0}=1$ when $B_0^\star=\emptyset$. The stage-1 $p$-value $P^{(1)}_{B_0}$ is $\mathcal F_1$-measurable. It is valid for $H_{B_0}$ by (ii), (iv) and Lemma~\ref{lem:comb}, because every $P^{(1)}_{a,k}$ that enters it tests a true hypothesis. Given $\mathcal F_1$, the selected dose $a^\star$ and hence the set $B_0^\star$ are fixed. On $\{B_0^\star\ne\emptyset\}$ every $H_{a^\star,k}$ with $k\in B_0^\star$ is true, so by (i) and (iii) the $P^{(2)}_k$, $k\in B_0^\star$, are conditionally valid given $\mathcal F_1$, and by (iv) and Lemma~\ref{lem:comb} so is $P^{(2)}_{B_0}$. On $\{B_0^\star=\emptyset\}$, $P^{(2)}_{B_0}=1$ is conditionally valid. Lemma~\ref{lem:inv} gives $\Pr\big(\cC(P^{(1)}_{B_0},P^{(2)}_{B_0})\le\alpha\big)\le\alpha$.
\end{proof}

The proof uses the selection rule, the futility rule and $n_2$ only through their $\mathcal F_1$-measurability. The closure over all $MK$ pairs makes $H_{B_0}$ a fixed hypothesis: it contains the null endpoints of every dose, so $P^{(1)}_{B_0}$ is one valid $p$-value whatever dose is selected.

\subsection{The dose-level closure}\label{app:union}

The dose-level closure of Section~\ref{sec:final} tests the hypotheses $H^\cup_a$, $a=1,\dots,M$, by closed testing over doses. Each intersection $H^\cup_I=\bigcap_{a\in I}H^\cup_a$, $I\ni a^\star$, is rejected if and only if $\cC\big(\psi_{|I|}(P^{(1)}_{a^\star,k},(Q_a)_{a\in I\setminus\{a^\star\}}),\,P^{(2)}_k\big)\le\alpha$ for every $k$, with $Q_a=\max_kP^{(1)}_{a,k}$.

\begin{proposition}\label{prop:union}
Under (i)--(iii) of Theorem~\ref{thm:t1e}, and (iv) for the stage-1 $p$-values, the dose-level closure satisfies $\Pr(\text{claim for }a^\star\text{ and }H^\cup_{a^\star}\text{ true})\le\alpha$ for every configuration.
\end{proposition}

\begin{proof}
Let $A_0=\{a:H^\cup_a\text{ true}\}$ and fix, for each $a\in A_0$, an endpoint $k_a$ with $H_{a,k_a}$ true. On $\{a^\star\notin A_0\}$ no claim is false. As in the proof of the theorem, pass to the trial that always proceeds. On $\{a^\star=a\}$ with $a\in A_0$, the claim requires the rejection of $H^\cup_{A_0}$, hence $\cC\big(\psi_{|A_0|}(P^{(1)}_{a,k_a},(Q_{a'})_{a'\in A_0\setminus\{a\}}),\,P^{(2)}_{k_a}\big)\le\alpha$. Since $Q_{a'}\ge P^{(1)}_{a',k_{a'}}$ and $\psi$ and $\cC$ are non-decreasing in each argument, this implies $\cC(P_1,P_2)\le\alpha$ with $P_1=\psi_{|A_0|}\big((P^{(1)}_{a',k_{a'}})_{a'\in A_0}\big)$ and with $P_2=P^{(2)}_{k_{a^\star}}$ on $\{a^\star\in A_0\}$ and $P_2=1$ otherwise. $P_1$ does not depend on the selected dose, is $\mathcal F_1$-measurable and is valid for the true intersection $\bigcap_{a'\in A_0}H_{a',k_{a'}}$ by Lemma~\ref{lem:comb}. $P_2$ is conditionally valid given $\mathcal F_1$, because $H_{a^\star,k_{a^\star}}$ is true on $\{a^\star\in A_0\}$. Lemma~\ref{lem:inv} completes the proof.
\end{proof}

The dose-level closure needs no Simes inequality at stage 2 and closes over $M$ doses instead of $MK$ pairs, which makes it more powerful than the full closure (Table~\ref{tab:closure}). The intersection--union $p$-values $Q_a$ (Lemma~\ref{lem:iu}) make its stage-1 component valid whichever dose is selected. Replacing them by the same-endpoint $p$-values $P^{(1)}_{a',k_{a^\star}}$ gives the per-endpoint closure, which is valid only if the other doses are also null on $k_{a^\star}$.

\subsection{Why the per-endpoint closure fails}\label{app:counter}

Suppose each endpoint is closed separately over doses, so that the intersection over doses $I\ni a^\star$ on endpoint $k$ is tested by $\cC\big(\psi_{|I|}((P^{(1)}_{a,k})_{a\in I}),\,P^{(2)}_k\big)$. Consider a configuration in which every dose is null on exactly one endpoint, dose $a$ on endpoint $k_a$, with $k_a\ne k_b$ for $a\ne b$ and effects on the other endpoints large enough that the corresponding $p$-values are close to zero, as in N3 and N4 of Section~\ref{sec:sim}. Then every claim is false. For the selected dose $a$ and its null endpoint $k_a$, every intersection $I\ni a$ with $|I|\ge2$ contains a dose $b\ne a$ whose $P^{(1)}_{b,k_a}$ is close to zero, so $\psi_{|I|}$ is close to zero and the intersection is rejected on endpoint $k_a$ with probability close to one. The binding condition is the elementary one, and
\begin{equation*}
\Pr(\text{false claim})\approx\sum_{a=1}^M\Pr\big(a^\star=a,\ \cC(P^{(1)}_{a,k_a},P^{(2)}_{k_a})\le\alpha\big).
\end{equation*}
If selection were independent of the stage-1 statistics of the null endpoints, each term would equal $\Pr(a^\star=a)\,\alpha$ and the sum would be $\alpha$. Selection is not independent of them. The composite gate favours a dose through the average of its $K$ endpoint contrasts, of which the null endpoint is one. The co-primary gate favours it through the minimum, which in this configuration is the null endpoint itself. Under the normal approximation the stage-1 statistic $Z^{(1)}_{a,k_a}$ and the event $\{a^\star=a\}$ are positively associated, so each term exceeds $\Pr(a^\star=a)\,\alpha$ and the sum exceeds $\alpha$. For the uniform composite with $K$ independent endpoints of equal variance, the correlation between $Z^{(1)}_{a,k_a}$ and the composite estimate $K^{-1}\sum_k\hat p_{a,k}$ of dose $a$, which ranks it against the other doses, is $1/\sqrt{2K}$, about $0.35$ for $K=4$. For the co-primary gate the ranking statistic of dose $a$ is nearly $Z^{(1)}_{a,k_a}$ itself, which explains its larger inflation in Table~\ref{tab:closure}. In the full closure the binding intersection is $H_{B_0}$ with $B_0=\{(1,k_1),\dots,(M,k_M)\}$, whose stage-1 $p$-value combines $M$ nearly uniform $p$-values. In the dose-level closure the other doses enter through $Q_b\ge P^{(1)}_{b,k_b}$, which is nearly uniform, with the same effect.

\section{Stage-1 sample-size ratio under parallel effects}\label{app:bracket}

\subsection{Consistency and concordance}

As $n_1\to\infty$ the posterior means converge to the true probabilities and the posterior variances are of order $1/n_1$, so the composite $z$-statistic satisfies $z^{\mathrm C}_a/\sqrt{n_1}\to(\bw^\top\bm\theta_a-\gamma)/s_a$. Here $s_a^2$ is the limit of $n_1\Var(\Delta_a\mid\bx)$, the sum over arms $a$ and $0$ of the subject-level variance of $\sum_kw_kY_k$. The composite gate therefore selects, with probability tending to one, the dose that maximises $(\bw^\top\bm\theta_a-\gamma)/s_a$. When the $s_a$ are equal across doses this is $a^{\mathrm C}=\arg\max_a\bw^\top\bm\theta_a$. The $s_a$ differ only through the effect of the dose on the variances $p_{a,k}(1-p_{a,k})$, by less than two per cent between any two doses in Section~\ref{sec:sim}, and the two maximisers coincide for every configuration, correlation and weight vector used there. Concordance, $a^{\mathrm C}=a^\circ$, is thus necessary for the PCS to tend to one, and under discordance the PCS tends to zero.

\subsection{The bracket}

Assume parallel effects $\theta_{a,k}=\theta_a$, $\delta_k=0$ and uniform weights. Assume also that $\hat p_{a,k}=p_{a,k}+\varepsilon_{a,k}$ for each arm $a$, where $\bm\varepsilon_a$ is a mean-zero normal vector with common variance $\sigma^2/n_1$ and common correlation $\phi$, independent across arms. Write $\varepsilon_{a,k}=\sigma n_1^{-1/2}\big(\sqrt\phi\,U_a+\sqrt{1-\phi}\,\xi_{a,k}\big)$ with $U_a,\xi_{a,1},\dots,\xi_{a,K}$ independent standard normal.

For the composite, $T^{\mathrm C}_a-T^{\mathrm C}_b=K^{-1}\sum_k(\hat p_{a,k}-\hat p_{b,k})$ has mean $\theta_a-\theta_b$ and variance $2\sigma^2n_1^{-1}\{\phi+(1-\phi)/K\}$. For the co-primary gate, $T^{\mathrm M}_a=\min_k(\hat p_{a,k}-\hat p_{0,k})$. Without the control error, $T^{\mathrm M}_a=\theta_a+\sigma n_1^{-1/2}\big(\sqrt\phi\,U_a+\sqrt{1-\phi}\min_k\xi_{a,k}\big)$. The mean of $\min_k\xi_{a,k}$ is common to all doses and cancels, so $\E[T^{\mathrm M}_a-T^{\mathrm M}_b]=\theta_a-\theta_b$ and $\Var[T^{\mathrm M}_a-T^{\mathrm M}_b]=2\sigma^2n_1^{-1}\{\phi+(1-\phi)V_K\}$ with $V_K=\Var[\min_k\xi_{a,k}]$. With the control error restored, $T^{\mathrm M}_a-T^{\mathrm M}_b$ acquires the term $-(e_{k_a}-e_{k_b})$, where $k_a$ and $k_b$ are the endpoints that attain the two minima and $e_k$ is the control error on endpoint $k$. This term vanishes when $k_a=k_b$ and otherwise adds variance of order $2\sigma^2n_1^{-1}(1-\phi)$, which moves the factor $\phi+(1-\phi)V_K$ towards $\phi+(1-\phi)$, the value for a single endpoint. Under the normal approximation the PCS is a function of the ratio of the pairwise signal $\theta_a-\theta_b$ to the pairwise standard deviation, so two gates reach the same PCS when $n_1$ is proportional to their variance factors. The ratio $n_1^{\mathrm C}/n_1^{\mathrm M}$ is then $\{\phi+(1-\phi)/K\}/\{\phi+(1-\phi)f\}$ with $f$ between $V_K$ and $1$, which is \eqref{eq:bracket}.

The minimum $\xi_{(1)}$ of $K$ independent standard normal variables has density $K\varphi(x)\{1-\Phi(x)\}^{K-1}$, and $V_K=\E[\xi_{(1)}^2]-\E[\xi_{(1)}]^2$. In the simulation study $\phi$ is evaluated at the mean control rate and a zero dose effect. The success probabilities there range from $0.43$ to $0.65$ across arms and endpoints, so the equal-variance assumption holds approximately.
